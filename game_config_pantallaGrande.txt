function actualizarIconosInventario() {
  inventario = inventario.filter((item) => item.cantidad > 0);

  for (let i = 0; i < inventarioStock; i++) {
    const slotIcon = iconosSlots[i];
    const texto = textosCantidad[i];
    const item = inventario[i];

    if (!item) {
      slotIcon.setVisible(false);
      texto.setVisible(false);
      continue;
    }

    slotIcon.setTexture(item.icono);
    slotIcon.setVisible(true);

    texto.setText(item.cantidad);
    texto.setVisible(true);
  }
}

function actualizarIconosPanel() {
  for (let i = 0; i < inventarioStock; i++) {
    const item = inventario[i];
    const slotIconInv = iconosSlotsInv[i];
    const textoCantidadInv = textosCantidadInv[i];

    if (!item) {
      slotIconInv.setVisible(false);
      textoCantidadInv.setVisible(false);
      continue;
    }

    slotIconInv.setTexture(item.icono);
    slotIconInv.setVisible(true);

    textoCantidadInv.setText(item.cantidad);
    textoCantidadInv.setVisible(true);
  }
}

function guardarEnInventario(nombre, tipo, cantidad, icono) {
  let itemEnInventario = inventario.find((item) => item.nombre === nombre);

  if (itemEnInventario) {
    itemEnInventario.cantidad += cantidad;
    return;
  }

  if (inventario.length >= inventarioStock) return;

  if (!itemEnInventario) {
    inventario.push({
      nombre: nombre,
      tipo: tipo,
      cantidad: cantidad,
      icono: icono,
    });
  }

  actualizarIconosInventario();
  actualizarIconosPanel();
}

function creacionArboles(scene) {
  for (let i = 0; i < 1900000; i++) {
    let x = Phaser.Math.Between(0, 10040);
    let y = Phaser.Math.Between(0, 10040);

    let valido = true;

    for (let p of posicionesOcupadas) {
      let dist = Phaser.Math.Distance.Between(x, y, p.x, p.y);
      if (dist < 100) {
        valido = false;
        break;
      }
    }

    if (valido) {
      posicionesOcupadas.push({ x, y });

      let arbol = scene.physics.add.staticSprite(x, y, "arbol");

      arbol.setOrigin(0.5, 1);
      arbol.setDepth(arbol.y);

      arboles.add(arbol);

      arbol.refreshBody();

      arbol.setSize(50, 10);
      arbol.setOffset(20, 133);

      return arbol;
    }
  }

  return null;
}

function creacionPiedras(scene) {
  for (let i = 0; i < 3500; i++) {
    let x = Phaser.Math.Between(0, 10040);
    let y = Phaser.Math.Between(0, 10040);

    let valido = true;

    for (let p of posicionesOcupadas) {
      let dist = Phaser.Math.Distance.Between(x, y, p.x, p.y);
      if (dist < 200) {
        valido = false;
        break;
      }
    }

    if (valido) {
      posicionesOcupadas.push({ x, y });

      let piedra = scene.physics.add.staticSprite(x, y, "icon_piedra");

      piedra.setDepth(piedra.y);

      piedras.add(piedra);

      piedra.refreshBody();

      piedra.setSize(6, 6);
      piedra.setOffset(10, 0);

      return piedra;
    }
  }

  return null;
}

function actualizarBarraEnergia() {
  if (this.energiaActual >= 0) {
    const porcentaje = this.energiaActual / this.energiaMax;

    this.barraEnergiaFondo.clear();
    this.barraEnergiaRelleno.clear();

    this.barraEnergiaFondo.fillStyle(0x000000, 0.5);
    this.barraEnergiaFondo.fillRect(20, 20, 200, 20);

    this.barraEnergiaRelleno.fillStyle(0x00ff00, 1);
    this.barraEnergiaRelleno.fillRect(20, 20, 200 * porcentaje, 20);

    if (this.text_stats_energia) {
      this.text_stats_energia.setText("Energia: " + this.energiaActual);
    }
  }
}

function puedeCraftear(receta) {
  const itemCraft = recetas.find((i) => i.resultado === receta);

  for (let ing of itemCraft.ingredientes) {
    const item = inventario.find((i) => i.nombre === ing.nombre);

    if (!item || item.cantidad < ing.cantidad) {
      return false;
    }
  }
  return true;
}

function craftear(receta) {
  if (!puedeCraftear(receta)) return "no se puede craftear";

  const itemCraft = recetas.find((i) => i.resultado === receta);

  itemCraft.ingredientes.forEach((ing) => {
    const item = inventario.find((i) => i.nombre === ing.nombre);
    item.cantidad -= ing.cantidad;
  });

  guardarEnInventario(
    itemCraft.resultado,
    itemCraft.tipo,
    itemCraft.cantidad,
    itemCraft.icon,
  );

  actualizarIconosInventario();
  actualizarIconosPanel();

  return "item crafteados";
}

        .setVisible(true);

      textosCantidadInv.push(txt_slot_inv);
    }
  }

  //INVENTARIO ZONA CRAFT

  const panel_craft_inv = this.add.graphics();
  panel_craft_inv.fillStyle(0x000011, 1);
  panel_craft_inv.fillRect(width / 2, 80, width / 2, height - 80);

  const panel_info_craft = this.add.graphics();
  panel_info_craft.fillStyle(0x001100, 1);
  panel_info_craft.fillRect(width / 2, 80, width / 2, height / 3);

  const slot_craft_info = this.add
    .image(width / 2 + 100, 200, "slot_image")
    .setScale(5);
  const icon_craft_info = this.add
    .image(width / 2 + 100, 200, "slot_image")
    .setScale(1.5)
    .setVisible(false);

  const titulo_craft_info = this.add.text(
    width / 2 + 170,
    150,
    "Item a Craftear",
    {
      fontSize: "20px",
      color: "#ffffff",
      stroke: "#000000",
      strokeThickness: 4,
    },
  );

  const text_craft_info = this.add.text(
    width / 2 + 170,
    190,
    "Presiona lo que quieres craftear,\n te daremos informacion sobre el objeto",
    {
      fontSize: "12px",
      color: "#ffffff",
      stroke: "#000000",
      strokeThickness: 4,
    },
  );

  const btn_craftear_item = this.add
    .text(width - 100, 200, "Craftear", {
      fontSize: "15px",
      color: "#ffffff",
      stroke: "#000000",
      strokeThickness: 4,
    })
    .setScrollFactor(0)
    .setInteractive();

  btn_craftear_item.on("pointerdown", () => {
    craftear(receta_a_craftear);
  });

  this.container_user_inv.add([
    panel_user_inv,
    panel_craft_inv,
    panel_info_craft,
    slot_craft_info,
    icon_craft_info,
    titulo_craft_info,
    text_craft_info,
    btn_craftear_item,
    ...SlotsInv,
    ...iconosSlotsInv,
    ...textosCantidadInv,
    titulo_inventario_personaje,
    titulo_inventario_maleta,
  ]);

  recetas.forEach((receta, i) => {
    let slotReceta = this.add
      .image(width / 2 + (50 + (i + i) * 36), height / 2, "slot_image")
      .setScale(3);
    let iconReceta = this.add
      .image(width / 2 + (50 + (i + i) * 36), height / 2, receta.icon)
      .setScale(1)
      .setInteractive()
      .setScrollFactor(0);

    iconReceta.on("pointerdown", () => {
      icon_craft_info.setTexture(receta.icon);
      icon_craft_info.setVisible(true);
      titulo_craft_info.setText("Fogata:");

      let textoIngrediente = "";

      receta.ingredientes.forEach((ingrediente) => {
        textoIngrediente += `- ${ingrediente.nombre} : ${ingrediente.cantidad}\n`;
      });

      text_craft_info.setText(textoIngrediente);
      receta_a_craftear = receta.resultado;
    });

    this.container_user_inv.add(slotReceta);
    this.container_user_inv.add(iconReceta);
  });

  //
  const panel_user_todo = this.add.graphics();
  panel_user_todo.fillStyle(0x110000, 1);
  panel_user_todo.fillRect(0, 80, width, height - 80);

  this.container_user_todo = this.add.container(0, 0, [panel_user_todo]);
  this.container_user_todo.setVisible(false);

  //
  const panel_user_config = this.add.graphics();
  panel_user_config.fillStyle(0x110000, 1);
  panel_user_config.fillRect(0, 80, width, height - 80);

  this.container_user_config = this.add.container(0, 0, [panel_user_config]);
  this.container_user_config.setVisible(false);

  this.showPanel = (panel) => {
    this.container_user_stats.setVisible(panel === "stats");
    this.container_user_inv.setVisible(panel === "inv");
    this.container_user_todo.setVisible(panel === "todo");
    this.container_user_config.setVisible(panel === "config");
  };

  text_user_stats.on("pointerdown", () => this.showPanel("stats"));
  text_user_inv.on("pointerdown", () => {
    this.showPanel("inv");
    actualizarIconosPanel();
  });
  text_user_todo.on("pointerdown", () => this.showPanel("todo"));
  text_user_confi.on("pointerdown", () => this.showPanel("config"));

  this.panelUserContainer.add([
    panel_user_bg,
    text_user_stats,
    text_user_inv,
    text_user_todo,
    text_user_confi,
    this.container_user_stats,
    this.container_user_inv,
    this.container_user_todo,
    this.container_user_config,
  ]);

  this.input.keyboard.on("keydown-E", () => {
    this.panel_estado_visible = !this.panel_estado_visible;
    this.showPanel("stats");
  });

  this.btn_salir_panel = this.add
    .image(window.innerWidth - 30, 30, "btn_salir_panel")
    .setInteractive()
    .setDepth(10002)
    .setScrollFactor(0)
    .setScale(2)
    .setVisible(false);
  this.btn_salir_panel.on("pointerdown", () => {
    this.panel_estado_visible = false;
  });

  window.addEventListener("resize", () => {
    game.scale.resize(window.innerWidth, window.innerHeight);
  });
}

function update() {
  let moving = false;
  player.setVelocity(0);

  player.setDepth(player.y);

  if (this.teclas.A.isDown || this.leftPressed) {
    player.setVelocityX(-100);
    vision_player = "oeste";
    player.play("caminar_oeste", true);
    moving = true;
  } else if (this.teclas.D.isDown || this.rightPressed) {
    player.setVelocityX(100);
    vision_player = "este";
    player.play("caminar_este", true);
    moving = true;
  } else if (this.teclas.W.isDown || this.upPressed) {
    player.setVelocityY(-100);
    vision_player = "norte";
    player.play("caminar_norte", true);
    moving = true;
  } else if (this.teclas.S.isDown || this.downPressed) {
    player.setVelocityY(100);
    vision_player = "sur";
    player.play("caminar_sur", true);
    moving = true;
  }

  if (!moving) {
    switch (vision_player) {
      case "norte":
        player.setTexture("player_parado_norte");
        break;
      case "sur":
        player.setTexture("player_parado_sur");
        break;
      case "este":
        player.setTexture("player_parado_este");
        break;
      case "oeste":
        player.setTexture("player_parado_oeste");
        break;
    }

    player.anims.stop();
  }

  //INVENTARIO
  if (slotSeleccionado == 0) {
    itemEnMano = inventario[slotSeleccionado]?.nombre || "nada";
  }
  if (slotSeleccionado == 1) {
    itemEnMano = inventario[slotSeleccionado]?.nombre || "nada";
  }
  if (slotSeleccionado == 2) {
    itemEnMano = inventario[slotSeleccionado]?.nombre || "nada";
  }
  if (slotSeleccionado == 3) {
    itemEnMano = inventario[slotSeleccionado]?.nombre || "nada";
  }
  if (slotSeleccionado == 4) {
    itemEnMano = inventario[slotSeleccionado]?.nombre || "nada";
  }

  //BOTON GOLPEAR
  if (this.btnGolpearDown || this.teclas.F.isDown) {
    if (!this.puedeGolpear) return;

    this.puedeGolpear = false;

    this.time.delayedCall(900, () => {
      this.puedeGolpear = true;
    });

    if (itemEnMano == "hacha") {
      arboles.children.iterate((arbol) => {
        if (!arbol) return;

        if (
          Phaser.Math.Distance.Between(player.x, player.y, arbol.x, arbol.y) <
          50
        ) {
          arbol.play("arbol_moviendose_golpe", true);

          this.energiaActual -= 1;
          this.actualizarBarraEnergia();

          if (arbol.vida == undefined) arbol.vida = 80;
          arbol.vida -= 10;

          if (arbol.vida <= 0) {
            arbol.play("arbol_cayendose_anim", true);
            arbol.setOrigin(0.5, 1);
            arbol.x += 61;
            this.time.delayedCall(1600, () => {
              arbol.destroy();
              guardarEnInventario("madera", "recurso", 5, "icon_madera");
              guardarEnInventario("hoja", "recurso", 5, "icon_hoja");
              actualizarIconosInventario();
              actualizarIconosPanel;
            });
          }
        }
      });
    }
  }

  if (this.btnInteractuarDown || this.teclas.Q.isDown) {
    if (!this.puedeInteractuar) return;

    this.puedeInteractuar = false;

    this.time.delayedCall(90, () => {
      this.puedeInteractuar = true;
    });

    if (inventario.length > inventarioStock) return;

    piedras.children.iterate((piedra) => {
      if (!piedra) return;

      if (
        Phaser.Math.Distance.Between(player.x, player.y, piedra.x, piedra.y) <
        50
      ) {
        this.energiaActual -= 1;
        this.actualizarBarraEnergia();

        piedra.destroy();

        this.guardarEnInventario("piedra", "recurso", 1, "icon_piedra");
        actualizarIconosInventario();
        actualizarIconosPanel();
      }
    });

    const item = inventario[slotSeleccionado];

    if (!item) return;

    if (item.tipo == "estructura") {
      const estructura = this.physics.add
        .staticImage(
          player.x + 15,
          player.y,
          inventario[slotSeleccionado].icono,
        )
        .setOrigin(0.5, 0.5);
      estructura.setDepth(estructura.y);
      estructura.setSize(30, 30);
      this.physics.add.collider(player, estructura);

      item.cantidad--;

      actualizarIconosInventario();
      actualizarIconosPanel();
    }
  }

  if (this.panel_estado_visible) {
    this.panelUserContainer.setVisible(true);
    this.btn_salir_panel.setVisible(true);
  } else {
    this.panelUserContainer.setVisible(false);
    this.btn_salir_panel.setVisible(false);
  }
}
